  *
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1"
LIMIT
  10;

-----------------------------------------------------------------------------------

SELECT DISTINCT state
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";

-----------------------------------------------------------------------------------
-- checking vehicle makes and are 97 

SELECT DISTINCT make
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";
-----------------------------------------------------------------------------------
-- first date of transaction
SELECT MIN(saledate) AS first_operating_date
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";

-----------------------------------------------------------------------------------
-- last date of transaction
SELECT MAX(saledate) AS last_operating_date
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";

-----------------------------------------------------------------------------------
-- number of units sold
SELECT COUNT(*) AS units_sold
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";

-----------------------------------------------------------------------------------
-- total revenue 
SELECT SUM(sellingprice) AS total_revenue
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1";

-------------------------------------------------------------
-- makes and models that generate the most revenue

SELECT make, model,
  SUM(sellingprice) AS total_revenue,
  COUNT(*) AS units_sold,
  ROUND(AVG(sellingprice), 2) AS avg_price
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1",
GROUP BY make, model
ORDER BY total_revenue DESC
LIMIT 20;

-------------------------------------------------------------
-- states with the highest sales volumes
SELECT state,
    COUNT(*) AS units_sold,
    SUM(sellingprice) AS total_revenue
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1",
GROUP BY state
ORDER BY units_sold DESC
LIMIT 10;

-------------------------------------------------------------
-- emerging trends in customer purchasing transmission preferences
SELECT
    CASE
    WHEN LOWER(transmission) LIKE 'auto%' THEN 'Automatic'
    WHEN LOWER(transmission) LIKE 'manu%' THEN 'Manual'
    ELSE 'Other'
    END AS transmission_type,
    COUNT(*) AS unit_sold
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1"
  GROUP BY transmission_type
  ORDER BY unit_sold DESC;

-------------------------------------------------------------
-- top selling car by shape
SELECT
    CASE 
    WHEN LOWER(body) LIKE '%suv%' THEN 'SUV'
    WHEN LOWER(body) LIKE '%sedan%' THEN 'Sedan'
    WHEN LOWER(body) LIKE '%hatchback%' THEN 'Hatchback'
    WHEN LOWER(body) LIKE '%pickup%'
    OR LOWER(body) LIKE '%truck%'
    OR LOWER(body) LIKE '%cab%' THEN 'Pickup'
ELSE 'Other'
END AS car_shape,
COUNT(*) AS units_sold
FROM
  "CAR_SALES"."CARS_ONE"."CAR_SALES1"
  GROUP BY car_shape
  ORDER BY units_sold DESC;

------------------------------------------------------------
-- analysing days and hours of operating

SELECT
    TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS') AS sale_ts,
    DAYNAME(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) AS day_name,

    CASE 
        WHEN DAYNAME(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) IN ('Saturday','Sunday')
            THEN 'weekend'
        ELSE 'weekday'
    END AS day_classification,

    MONTHNAME(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) AS month_name,

    CASE
        WHEN HOUR(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) BETWEEN 6 AND 11 THEN 'Morning'
        WHEN HOUR(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) BETWEEN 12 AND 16 THEN 'Afternoon'
        WHEN HOUR(TO_TIMESTAMP_NTZ(saledate, 'DY MON DD YYYY HH24:MI:SS')) >= 17 THEN 'Evening'
        ELSE 'Overnight'
    END AS time_bucket,

    YEAR,
    make,
    model,
    trim,
    body,
    transmission,
    vin,
    state,
    condition,
    odometer,
    color,
    interior,
    seller,
    mmr,

    SUM(TRY_TO_NUMBER(sellingprice)) AS total_revenue,
    COUNT(*) AS units_sold,
    ROUND(AVG(TRY_TO_NUMBER(sellingprice)), 2) AS avg_price

FROM "CAR_SALES"."CARS_ONE"."CAR_SALES1"

GROUP BY
    sale_ts,
    day_name,
    day_classification,
    month_name,
    time_bucket,
    YEAR, make, model, trim, body, transmission, vin, state,
    condition, odometer, color, interior, seller, mmr

ORDER BY month_name, day_name, time_bucket;
